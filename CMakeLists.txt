cmake_minimum_required(VERSION 3.24)
project(Saturday LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SATURDAY_FETCH_DEPS "Download pinned HTTP/JSON/HTML dependencies" ON)
option(SATURDAY_LLAMA "Build experimental direct llama.cpp backend" OFF)
set(MYSQL_ROOT "" CACHE PATH "MySQL Server or C client installation")
find_package(Threads REQUIRED)
if(SATURDAY_FETCH_DEPS)
  include(FetchContent)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(BUILD_CURL_EXE OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  set(CURL_USE_LIBPSL OFF CACHE BOOL "" FORCE)
  set(CURL_ZLIB OFF CACHE BOOL "" FORCE)
  set(CURL_BROTLI OFF CACHE BOOL "" FORCE)
  set(CURL_ZSTD OFF CACHE BOOL "" FORCE)
  set(CURL_USE_LIBSSH2 OFF CACHE BOOL "" FORCE)
  if(WIN32)
    set(CURL_USE_SCHANNEL ON CACHE BOOL "" FORCE)
    set(CURL_USE_OPENSSL OFF CACHE BOOL "" FORCE)
  endif()
  set(SIMDJSON_DEVELOPER_MODE OFF CACHE BOOL "" FORCE)
  set(YYJSON_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(LEXBOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(LEXBOR_BUILD_SHARED OFF CACHE BOOL "" FORCE)
  set(LEXBOR_BUILD_STATIC ON CACHE BOOL "" FORCE)
  FetchContent_Declare(curl URL https://github.com/curl/curl/releases/download/curl-8_14_1/curl-8.14.1.tar.xz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_Declare(simdjson URL https://github.com/simdjson/simdjson/archive/refs/tags/v3.12.3.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_Declare(yyjson URL https://github.com/ibireme/yyjson/archive/refs/tags/0.12.0.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_Declare(lexbor URL https://github.com/lexbor/lexbor/archive/refs/tags/v2.5.0.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(curl simdjson yyjson lexbor)
else()
  find_package(CURL REQUIRED)
  find_package(simdjson CONFIG REQUIRED)
  find_package(yyjson CONFIG REQUIRED)
  find_package(lexbor CONFIG REQUIRED)
endif()
find_path(MYSQL_INCLUDE_DIR mysql.h HINTS "${MYSQL_ROOT}/include" "C:/Program Files/MySQL/MySQL Server 8.0/include" PATH_SUFFIXES mysql mariadb REQUIRED)
find_library(MYSQL_LIBRARY NAMES libmysql mysql mysqlclient mariadb HINTS "${MYSQL_ROOT}/lib" "C:/Program Files/MySQL/MySQL Server 8.0/lib" REQUIRED)
file(GLOB SOURCES CONFIGURE_DEPENDS "src/*.cpp")
if(NOT SATURDAY_LLAMA)
  list(FILTER SOURCES EXCLUDE REGEX "inference.cpp$")
endif()
add_executable(saturday main.cpp ${SOURCES})
target_include_directories(saturday PRIVATE include "${MYSQL_INCLUDE_DIR}")
if(TARGET lexbor_static)
  set(HTML_TARGET lexbor_static)
elseif(TARGET lexbor::lexbor)
  set(HTML_TARGET lexbor::lexbor)
else()
  set(HTML_TARGET lexbor)
endif()
if(TARGET yyjson::yyjson)
  set(JSON_TARGET yyjson::yyjson)
else()
  set(JSON_TARGET yyjson)
endif()
target_link_libraries(saturday PRIVATE CURL::libcurl simdjson::simdjson ${JSON_TARGET} ${HTML_TARGET} "${MYSQL_LIBRARY}" Threads::Threads)
if(WIN32)
  target_compile_definitions(saturday PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(saturday PRIVATE ws2_32 winmm shell32)
  get_filename_component(MYSQL_LIB_DIR "${MYSQL_LIBRARY}" DIRECTORY)
  if(EXISTS "${MYSQL_LIB_DIR}/libmysql.dll")
    add_custom_command(TARGET saturday POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${MYSQL_LIB_DIR}/libmysql.dll" "$<TARGET_FILE_DIR:saturday>")
  endif()
else()
  find_package(ALSA REQUIRED)
  target_link_libraries(saturday PRIVATE ALSA::ALSA)
endif()
if(SATURDAY_LLAMA)
  find_path(LLAMA_INCLUDE_DIR llama.h REQUIRED)
  find_library(LLAMA_LIBRARY llama REQUIRED)
  target_include_directories(saturday PRIVATE "${LLAMA_INCLUDE_DIR}")
  target_link_libraries(saturday PRIVATE "${LLAMA_LIBRARY}")
endif()
if(MSVC)
  target_compile_options(saturday PRIVATE /W4 /utf-8)
else()
  target_compile_options(saturday PRIVATE -Wall -Wextra)
endif()
enable_testing()
add_executable(platform_tests tests/platform_tests.cpp src/text_stream.cpp src/config.cpp src/client.cpp src/client_windows.cpp)
target_include_directories(platform_tests PRIVATE include)
target_link_libraries(platform_tests PRIVATE Threads::Threads)
if(WIN32)
  target_compile_definitions(platform_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(platform_tests PRIVATE ws2_32)
endif()
add_test(NAME platform_tests COMMAND platform_tests)

set_tests_properties(platform_tests PROPERTIES TIMEOUT 30)

add_executable(core_tests tests/core_tests.cpp ${SOURCES})
get_target_property(APP_INCLUDES saturday INCLUDE_DIRECTORIES)
get_target_property(APP_LINKS saturday LINK_LIBRARIES)
target_include_directories(core_tests PRIVATE ${APP_INCLUDES})
target_link_libraries(core_tests PRIVATE ${APP_LINKS})
if(WIN32)
  target_compile_definitions(core_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
endif()
add_test(NAME core_tests COMMAND core_tests)
set_tests_properties(core_tests PROPERTIES TIMEOUT 30)

target_compile_definitions(core_tests PRIVATE SATURDAY_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
